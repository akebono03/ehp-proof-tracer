def _render_unicode_math_name_latex(
  name: str,
) -> str:
  if not isinstance(
    name,
    str,
  ):
    raise TypeError(
      "name must be a str"
    )

  if not name:
    raise ValueError(
      "name must not be empty"
    )

  # Normalize exceptional DB labels without altering their typed identity.
  # The source HyoujiGen.csv stores these labels as Unicode, while their
  # mathematical TeX equivalents use nested bars and numeric subscripts.
  import re

  bar_pattern = re.compile(r"([Α-Ωα-ω])([\u0304\u0305]+)")

  def render_bars(match):
    base = _GREEK_LATEX.get(match.group(1), match.group(1))
    level = len(match.group(2))
    for _ in range(level):
      base = r"\bar{" + base + "}"
    return base

  name = bar_pattern.sub(render_bars, name)
  index_pattern = re.compile(r"([Α-Ωα-ω])([0-9]+)")
  name = index_pattern.sub(
    lambda match: match.group(1) + "".join(
      next(key for key, value in _SUBSCRIPT_DIGITS.items() if value == digit)
      for digit in match.group(2)
    ),
    name,
  )

  pieces: list[str] = []
  subscript_digits: list[str] = []
  superscript_digits: list[str] = []
  prime_count = 0

  def flush_scripts() -> None:
    nonlocal subscript_digits
    nonlocal superscript_digits

    if subscript_digits:
      pieces.append(
        "_{"
        + "".join(
          subscript_digits
        )
        + "}"
      )
      subscript_digits = []

    if superscript_digits:
      pieces.append(
        "^{"
        + "".join(
          superscript_digits
        )
        + "}"
      )
      superscript_digits = []

  for character in name:
    if character in _SUBSCRIPT_DIGITS:
      subscript_digits.append(
        _SUBSCRIPT_DIGITS[
          character
        ]
      )
      continue

    if character in _SUPERSCRIPT_DIGITS:
      superscript_digits.append(
        _SUPERSCRIPT_DIGITS[
          character
        ]
      )
      continue

    if character in _PRIME_CHARS:
      flush_scripts()

      if character == "″":
        prime_count += 2
      elif character == "‴":
        prime_count += 3
      else:
        prime_count += 1
      continue

    flush_scripts()

    pieces.append(
      _GREEK_LATEX.get(
        character,
        character,
      )
    )

  flush_scripts()

  if prime_count:
    pieces.append(
      "'" * prime_count
    )

  return "".join(
    pieces
  )
