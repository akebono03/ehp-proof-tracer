def _parse_integer_form_value(
  value: str | None,
  field_name: str,
) -> int:
  if value is None or not value.strip():
    raise ValueError(
      f"{field_name} is required"
    )

  try:
    return int(
      value
    )
  except ValueError as error:
    raise ValueError(
      f"{field_name} must be an integer"
    ) from error
