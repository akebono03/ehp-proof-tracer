def format_complete_proof_sentences(lines: list[str], model) -> list[str]:
    """Keep the original tree/citations; alter only completed narrative prose."""
    from phase174_r3_r7_r70_verified_narrative_projection.projection import (
        project_verified_general_narrative,
    )
    from phase175_r3r11_cited_hopf_narrative.projection import (
        project_verified_source_and_hopf_summary,
    )
    combined = project_verified_general_narrative(
        combine_adjacent_verified_eta_sentences(lines, model), model,
    )
    cited = project_verified_source_and_hopf_summary(combined, model)
    formatted = split_completed_proof_sentences(cited)
    from phase176_r3r56_cited_source_display_isolated.projection import (
        project_cited_sources,
    )
    markdown = "\n".join(formatted)
    projected = project_cited_sources(markdown, model)
    from phase176_r3r57_composite_hopf_display_isolated.projection import (
        project_composite_hopf_chain,
    )
    compressed, _ = project_composite_hopf_chain(projected, model)
    from phase176_r3r58_finite_cyclic_lift_display_isolated.projection import (
        project_finite_cyclic_lift,
    )
    explained, _ = project_finite_cyclic_lift(compressed, model)
    from phase176_r3r59r3_definitional_suspension_elision_isolated.projection import (
        project_definitional_suspension_elision,
    )
    elided, _ = project_definitional_suspension_elision(explained, model)
    return formatted if elided == markdown else elided.split("\n")
