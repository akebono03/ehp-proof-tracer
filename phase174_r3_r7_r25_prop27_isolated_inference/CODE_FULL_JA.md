```python
"""Isolated, source-first typed specialization of Toda Proposition 2.7.

No public registration, no GIVEN facts, and no self-issued certified premise.
"""
from __future__ import annotations

from dataclasses import dataclass

from homotopy_groups import TodaDeltaMap, TodaHopfInvariantMap, TodaPrimaryGroup
from proof import InferenceRule, LiteratureReference, PremisePattern, ProofRule


PROP27_REFERENCE = LiteratureReference(
    label="Toda Proposition 2.7",
    author="H. Toda",
    title="Composition Methods in Homotopy Groups of Spheres",
    locator="Proposition 2.7",
)
RULE_NAME = "Toda Proposition 2.7 even-dimensional typed Hopf-Delta specialization"


@dataclass(frozen=True)
class Prop27GeneralTheoremPremise:
    """Theorem-schema obligation, not itself a proved fact or a GIVEN."""
    source_locator: str = "Proposition 2.7"
    only_even_n: bool = True
    sign_undetermined: bool = True


@dataclass(frozen=True)
class Prop27TypedHopfDeltaStatement:
    n: int
    delta: TodaDeltaMap
    hopf: TodaHopfInvariantMap
    source_iota_dimension: int
    target_iota_dimension: int
    absolute_coefficient: int
    sign_undetermined: bool


def typed_specialization(n: int) -> Prop27TypedHopfDeltaStatement:
    """Build typed data only: this operation never creates a ProofStep."""
    if type(n) is not int or n < 2 or n % 2:
        raise ValueError("Proposition 2.7 requires an even integer n >= 2")
    source = TodaPrimaryGroup(2 * n + 1, 2 * n + 1)
    middle = TodaPrimaryGroup(2 * n - 1, n)
    target = TodaPrimaryGroup(2 * n - 1, 2 * n - 1)
    return Prop27TypedHopfDeltaStatement(
        n=n,
        delta=TodaDeltaMap(source, middle),
        hopf=TodaHopfInvariantMap(middle, target),
        source_iota_dimension=2 * n + 1,
        target_iota_dimension=2 * n - 1,
        absolute_coefficient=2,
        sign_undetermined=True,
    )


def prop27_typed_specialization_rule(n: int) -> InferenceRule:
    """Conditional inference from an independently established cited schema.

    The rule does not construct or register the necessary schema premise.
    """
    conclusion = typed_specialization(n)
    required = Prop27GeneralTheoremPremise()

    def guard(premises, bindings):
        if len(premises) != 1:
            return False
        step = premises[0]
        previous = step.inference_rule
        if step.rule is not ProofRule.INFERENCE or step.conclusion != required:
            return False
        if not isinstance(previous, InferenceRule):
            return False
        citation = previous.literature_reference
        return (
            isinstance(citation, LiteratureReference)
            and citation.locator == "Proposition 2.7"
            and citation.label == PROP27_REFERENCE.label
            and previous is not rule
        )

    rule = InferenceRule(
        name=RULE_NAME,
        description=(
            "If an independently sourced and inferred Toda Proposition 2.7 "
            "general theorem premise is present, specialize it at even n. "
            "The result retains both EHP map types and the undetermined sign. "
            "No source premise is generated by this rule."
        ),
        premise_patterns=(PremisePattern(
            statement_type=Prop27GeneralTheoremPremise,
            statement_pattern=required,
            proof_rule=ProofRule.INFERENCE,
        ),),
        conclusion_pattern=conclusion,
        match_guard=guard,
        literature_reference=PROP27_REFERENCE,
    )
    return rule
```

## `audit.py`

```python
"""R25 source-first guard with reviewed-ancestor lineage and explicit diagnostics.

R24 commits may follow the reviewed R23 merge; unrelated commits are rejected.
The single tracked pytest nodeids cache file may occur in reviewed commits.
No mathematics, production rule, GIVEN or search registration is changed.
"""
from __future__ import annotations

import argparse
from pathlib import Path
import subprocess

from phase174_r3_r7_r25_prop27_isolated_inference.rule import typed_specialization

ROOT = Path(__file__).resolve().parents[1]
REVIEWED_HEAD = "5349f850428085a162f5b286deebfaa0bdcfb9d0"
ALLOWED_TRACKED = {".pytest_cache/v/cache/nodeids"}
ALLOWED_COMMIT_PREFIXES = (
    "phase174_r3_r7_r24_prop27_general_typed_design/",
    "phase174_r3_r7_r25_prop27_isolated_inference/",
)
CHECKED = (
    "sources/toda/Toda_02_2.tex",
    "sources/toda/Toda_05_2.tex",
    "proof.py",
    "homotopy_groups.py",
    "tests/test_phase50_prop27_theorem_semantics.py",
)


def git(*args: str) -> str:
    return subprocess.run(
        ("git",) + args, cwd=ROOT, capture_output=True,
        text=True, check=True, encoding="utf-8", errors="replace",
    ).stdout.strip()


def is_ancestor(ancestor: str, descendant: str) -> bool:
    process = subprocess.run(
        ("git", "merge-base", "--is-ancestor", ancestor, descendant),
        cwd=ROOT, capture_output=True,
    )
    if process.returncode not in (0, 1):
        raise RuntimeError("Could not inspect Git ancestry")
    return process.returncode == 0


def classify_lineage(local: str, remote: str, has_reviewed_ancestor: bool,
                     changed_committed: tuple[str, ...]) -> str:
    if not has_reviewed_ancestor:
        raise RuntimeError("Local HEAD does not descend from reviewed HEAD")
    if remote not in (REVIEWED_HEAD, local):
        raise RuntimeError(
            "Remote develop is neither reviewed HEAD nor local HEAD; "
            "fetch and independently review its history"
        )
    bad = tuple(path for path in changed_committed
                if path not in ALLOWED_TRACKED and not path.startswith(ALLOWED_COMMIT_PREFIXES))
    if bad:
        raise RuntimeError("Unreviewed committed paths: " + repr(bad))
    return "REVIEWED_HEAD" if local == REVIEWED_HEAD else "R24_R25_ONLY_DESCENDANT"


def audit() -> tuple[str, str, str]:
    if Path(git("rev-parse", "--show-toplevel")).resolve() != ROOT.resolve():
        raise RuntimeError("Extract R25 at repository root")
    if git("branch", "--show-current") != "develop":
        raise RuntimeError("Expected develop branch")
    local = git("rev-parse", "HEAD")
    remote_output = git("ls-remote", "origin", "refs/heads/develop")
    if not remote_output:
        raise RuntimeError("Cannot read remote develop HEAD")
    remote = remote_output.split()[0]
    print("PHASE174_R3R7R25_LOCAL_HEAD=" + local)
    print("PHASE174_R3R7R25_REMOTE_HEAD=" + remote)
    changed_committed = tuple(git("diff", "--name-only", REVIEWED_HEAD, "HEAD").splitlines())
    lineage = classify_lineage(
        local, remote, is_ancestor(REVIEWED_HEAD, local), changed_committed,
    )
    modified = set(git("diff", "--name-only").splitlines()) | set(
        git("diff", "--cached", "--name-only").splitlines()
    )
    unexpected = modified - ALLOWED_TRACKED
    if unexpected:
        raise RuntimeError("Unreviewed tracked changes: " + repr(sorted(unexpected)))
    for path in CHECKED:
        stored = subprocess.run(
            ("git", "show", REVIEWED_HEAD + ":" + path), cwd=ROOT,
            capture_output=True, check=True,
        ).stdout
        actual = (ROOT / path).read_bytes()
        lf = stored.replace(b"\r\n", b"\n")
        if actual not in (stored, lf, lf.replace(b"\n", b"\r\n")):
            raise RuntimeError("Reviewed source changed: " + path)
    source = (ROOT / CHECKED[0]).read_text(encoding="utf-8-sig")
    if (r"\begin{proposition}\label{prop:2.7}" not in source
            or r"H\!\bigl(\Delta(\iota_{2n+1})\bigr)\ =\ \pm\,2\,\iota_{2n-1}" not in source
            or "$n$ が偶数のとき" not in source):
        raise RuntimeError("Proposition 2.7 source contract changed")
    value = typed_specialization(6)
    if value.delta.target_group != value.hopf.source_group:
        raise RuntimeError("Delta/Hopf domain mismatch")
    return local, remote, lineage


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--dry-run", action="store_true")
    args = parser.parse_args()
    local, remote, lineage = audit()
    prefix = "PHASE174_R3R7R25_"
    print(prefix + "SOURCE_GUARD=PASS")
    print(prefix + "HEAD=" + local)
    print(prefix + "LINEAGE=" + lineage)
    print(prefix + "N6_TYPED_DELTA_HOPF=True")
    print(prefix + "SIGNED_RELATION_UNDETERMINED=True")
    print(prefix + "NEW_GIVEN=0")
    print(prefix + "PUBLIC_REGISTRATION=False")
    print(prefix + "PUBLIC_VERIFIED_STEPS=NOT_CHECKED")
    print(prefix + "ROOT_COMPLETE=False")
    print(prefix + "FULL_PYTEST=False")
    print(prefix + "MODE=" + ("DRYRUN" if args.dry_run else "AUDIT"))
    print(prefix + "GATE=PASS_R25_ISOLATED_TYPED_INFERENCE_PUBLIC_PREMISE_PENDING")
    print("(END)")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

"""Focused R25 tests; fabricated premises are test fixtures, not public proof."""
from __future__ import annotations

import pytest

from homotopy_groups import TodaPrimaryGroup
from proof import (
    InferenceRule, LiteratureReference, ProofRule, ProofStep,
    find_inference_match, run_inference_until_stable_with_history,
)
from phase174_r3_r7_r25_prop27_isolated_inference.rule import (
    PROP27_REFERENCE, Prop27GeneralTheoremPremise,
    Prop27TypedHopfDeltaStatement, prop27_typed_specialization_rule,
    typed_specialization,
)


def fixture_step(*, locator="Proposition 2.7", rule=ProofRule.INFERENCE,
                 sign=True):
    parent = InferenceRule(
        name="test-only independent cited schema premise",
        literature_reference=LiteratureReference(
            label="Toda Proposition 2.7", locator=locator,
        ),
    )
    return ProofStep(
        conclusion=Prop27GeneralTheoremPremise(sign_undetermined=sign),
        premises=(), rule=rule, inference_rule=parent,
    )


def test_n6_correct_groups_and_sign():
    result = typed_specialization(6)
    assert isinstance(result, Prop27TypedHopfDeltaStatement)
    assert result.delta.source_group == TodaPrimaryGroup(13, 13)
    assert result.delta.target_group == TodaPrimaryGroup(11, 6)
    assert result.hopf.source_group == TodaPrimaryGroup(11, 6)
    assert result.hopf.target_group == TodaPrimaryGroup(11, 11)
    assert (result.source_iota_dimension, result.target_iota_dimension) == (13, 11)
    assert result.absolute_coefficient == 2
    assert result.sign_undetermined is True


@pytest.mark.parametrize("n", [0, 1, 3, 5, -2, 6.0, True, "6"])
def test_reject_invalid_dimensions(n):
    with pytest.raises(ValueError):
        typed_specialization(n)


def test_valid_even_generalization():
    for n in (2, 4, 6, 8, 10):
        value = typed_specialization(n)
        assert value.delta.target_group == value.hopf.source_group
        assert value.sign_undetermined


def test_no_premise_no_proof():
    assert find_inference_match(prop27_typed_specialization_rule(6), ()) is None


def test_given_premise_rejected():
    rule = prop27_typed_specialization_rule(6)
    assert find_inference_match(rule, (fixture_step(rule=ProofRule.GIVEN),)) is None


def test_wrong_source_rejected():
    rule = prop27_typed_specialization_rule(6)
    assert find_inference_match(rule, (fixture_step(locator="Proposition 5.9"),)) is None


def test_missing_sign_preservation_rejected():
    rule = prop27_typed_specialization_rule(6)
    assert find_inference_match(rule, (fixture_step(sign=False),)) is None


def test_synthetic_cited_fixture_produces_inference_not_given():
    premise = fixture_step()
    rule = prop27_typed_specialization_rule(6)
    result = run_inference_until_stable_with_history(rule, (premise,))
    derived = [s for s in result.steps if isinstance(s.conclusion, Prop27TypedHopfDeltaStatement)]
    assert len(derived) == 1
    assert derived[0].rule is ProofRule.INFERENCE
    assert derived[0].premises == (premise,)
    assert derived[0].inference_rule.literature_reference == PROP27_REFERENCE
    assert derived[0].conclusion == typed_specialization(6)


def test_rule_does_not_create_premise():
    rule = prop27_typed_specialization_rule(6)
    assert len(rule.premise_patterns) == 1
    assert rule.premise_patterns[0].statement_type is Prop27GeneralTheoremPremise
    assert rule.premise_patterns[0].proof_rule is ProofRule.INFERENCE
