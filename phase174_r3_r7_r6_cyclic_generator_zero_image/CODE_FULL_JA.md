~~~~python
"""Generic zero-image inference from a proved finite-cyclic generator and its image.

This is an isolated inference candidate. No statement is admitted as a GIVEN;
no production search provider or renderer is changed.
"""
from __future__ import annotations

from expression import Expression, Zero
from homotopy_groups import FiniteCyclicGroup, TodaPrimaryGroup, TodaSuspensionMap
from map_property_rules import ImageZeroStatement
from phase165_r14_r6_generic_iso import GeneratorImage
from phase165_r3_recursive_search import GoalExpansion
from proof import InferenceRule, PremisePattern, ProofRule, Relation, RelationType


RULE_NAME = "finite cyclic generator mapped to zero implies zero homomorphism image"


def cyclic_generator_zero_image_inputs(
    goal: object,
    presentation: object,
) -> tuple[Relation, GeneratorImage] | None:
    """Construct two unproved typed obligations without choosing a generator.

    The caller supplies a concrete, separately screened cyclic-presentation
    candidate. It remains a GOAL, not a verified group claim. Only a genuine
    suspension map with valid adjacent dimensions is used at this phase.
    """
    if not isinstance(goal, ImageZeroStatement):
        return None
    suspension = goal.map
    if not isinstance(suspension, TodaSuspensionMap):
        return None
    domain = suspension.source_group
    codomain = suspension.target_group
    if not isinstance(domain, TodaPrimaryGroup) or not isinstance(codomain, TodaPrimaryGroup):
        return None
    degree = domain.group_dimension
    sphere = domain.sphere_dimension
    if (type(degree) is not int or type(sphere) is not int
            or sphere < 2 or degree < sphere
            or codomain != TodaPrimaryGroup(degree + 1, sphere + 1)):
        return None
    if (not isinstance(presentation, Relation)
            or presentation.relation_type is not RelationType.EQUALITY
            or presentation.lhs != domain
            or not isinstance(presentation.rhs, FiniteCyclicGroup)):
        return None
    cyclic = presentation.rhs
    if (type(cyclic.order) is not int or cyclic.order < 2
            or not isinstance(cyclic.generator, Expression)
            or isinstance(cyclic.generator, Zero)):
        return None
    image_goal = GeneratorImage(
        map=suspension,
        source_generator=cyclic.generator,
        target_generator=Zero(),
    )
    return presentation, image_goal


def zero_image_from_cyclic_generator_rule(
    goal: ImageZeroStatement,
    presentation: Relation,
) -> InferenceRule:
    """Require independent inferred proofs of the presentation and E(g)=0."""
    required = cyclic_generator_zero_image_inputs(goal, presentation)
    if required is None:
        raise ValueError("invalid finite cyclic zero-image premises")

    def guard(premises, bindings):
        return (
            len(premises) == 2
            and all(step.rule is ProofRule.INFERENCE for step in premises)
            and tuple(step.conclusion for step in premises) == required
            and cyclic_generator_zero_image_inputs(goal, presentation) == required
            and isinstance(required[1].target_generator, Zero)
        )

    return InferenceRule(
        name=RULE_NAME,
        description=(
            "If a homomorphism has a separately proved finite-cyclic domain "
            "generated by g, and the independently inferred generator image "
            "is the typed zero element, every element maps to zero. "
            "Neither the domain presentation nor E(g)=0 is a new GIVEN."
        ),
        premise_patterns=(
            PremisePattern(
                proof_rule=ProofRule.INFERENCE,
                statement_type=Relation,
                relation_type=RelationType.EQUALITY,
                statement_pattern=required[0],
            ),
            PremisePattern(
                proof_rule=ProofRule.INFERENCE,
                statement_type=GeneratorImage,
                statement_pattern=required[1],
            ),
        ),
        conclusion_pattern=goal,
        match_guard=guard,
    )


def expand_zero_image_from_cyclic_generator(
    goal: object,
    presentation: object,
) -> tuple[GoalExpansion, ...]:
    """Return a backward candidate only; never assert a verified ProofStep."""
    required = cyclic_generator_zero_image_inputs(goal, presentation)
    if required is None:
        return ()
    return (GoalExpansion(
        rule=zero_image_from_cyclic_generator_rule(goal, presentation),
        subgoals=required,
    ),)
~~~~

~~~~python
def git(*args: str, cwd: Path = ROOT) -> str:
    result = subprocess.run(

def blob_sha(data: bytes) -> str:
    return hashlib.sha1(b"blob " + str(len(data)).encode("ascii") + b"\0" + data).hexdigest()


def accepted_source_blob(data: bytes, expected: str) -> bool:
    normalized = data.replace(b"\r\n", b"\n")
    return any(blob_sha(item) == expected for item in (
        data, normalized, normalized.replace(b"\n", b"\r\n"),

def validate_reviewed_head_lineage(
    local: str, remote: str, successor_commits: tuple[str, ...],
    changed_paths: tuple[str, ...],

def verify_git_and_sources(root: Path = ROOT) -> dict[str, Any]:
    if Path(git("rev-parse", "--show-toplevel", cwd=root)).resolve() != root.resolve():
        raise RuntimeError("Package must be installed directly under repository root")
    if git("branch", "--show-current", cwd=root) != "develop":
        raise RuntimeError("Expected develop branch")
    local = git("rev-parse", "HEAD", cwd=root)
    line = git("ls-remote", "origin", "refs/heads/develop", cwd=root)
    remote = line.split()[0] if line else ""
    if git("diff", "--cached", "--name-only", cwd=root):
        raise RuntimeError("Staged changes require review")
    if git("diff", "--name-only", cwd=root):
        raise RuntimeError("Tracked changes require review")
    if local == REVIEWED_HEAD:
        successor_commits: tuple[str, ...] = ()
        changed_paths: tuple[str, ...] = ()
    else:
        subprocess.run(
            ("git", "merge-base", "--is-ancestor", REVIEWED_HEAD, local),
            cwd=root, check=True, capture_output=True,
        )
        successor_commits = tuple(
            git("rev-list", "--reverse", f"{REVIEWED_HEAD}..{local}", cwd=root).splitlines()
        )
        changed_paths = tuple(
            git("diff", "--name-only", f"{REVIEWED_HEAD}..{local}", cwd=root).splitlines()
        )
    lineage = validate_reviewed_head_lineage(local, remote, successor_commits, changed_paths)
    for relative, expected in SOURCE_BLOBS.items():
        file = root / relative
        if not file.is_file() or not accepted_source_blob(file.read_bytes(), expected):
            raise RuntimeError("Pinned source changed: " + relative)

        "local_head": local,
        "remote_head": remote,
        "branch": "develop",
        "source_guard": "PASS",
        "head_lineage": lineage,
        "successor_changed_paths": list(changed_paths),
~~~~

~~~~python
def fixture_case(degree: int = 10, sphere: int = 5, order: int = 2):
    domain = TodaPrimaryGroup(degree, sphere)
    mapping = TodaSuspensionMap(domain, TodaPrimaryGroup(degree + 1, sphere + 1))
    generator = HomotopyElement("a", sphere, source=degree, target=sphere)
    presentation = Relation(
        lhs=domain,
        rhs=FiniteCyclicGroup(order=order, generator=generator),
        relation_type=RelationType.EQUALITY,
    )
    return ImageZeroStatement(mapping), presentation


def inferred(statement: object) -> ProofStep:
    return ProofStep(
        conclusion=statement, premises=(), rule=ProofRule.INFERENCE,
        inference_rule=InferenceRule(name="test-only independently inferred fixture"),
    )


@pytest.mark.parametrize("degree,sphere,order", (
    (10, 5, 2), (11, 6, 4), (20, 12, 8), (14, 7, 3),
))
def test_generic_valid_suspension_and_cyclic_presentation(
    degree: int, sphere: int, order: int,
) -> None:
    goal, presentation = fixture_case(degree, sphere, order)
    children = cyclic_generator_zero_image_inputs(goal, presentation)
    assert children is not None
    assert children[0] == presentation
    assert children[1] == GeneratorImage(
        goal.map, presentation.rhs.generator, Zero()
    )
    candidates = expand_zero_image_from_cyclic_generator(goal, presentation)
    assert len(candidates) == 1
    assert candidates[0].subgoals == children
    assert candidates[0].rule.name == RULE_NAME


def test_two_proved_premises_derive_zero_image_and_keep_ancestors() -> None:
    goal, presentation = fixture_case()
    children = cyclic_generator_zero_image_inputs(goal, presentation)
    assert children is not None
    proofs = tuple(inferred(child) for child in children)

    assert len(matches) == 1
    derived = derive_inference_proof_step(matches[0])
    assert derived.rule is ProofRule.INFERENCE
    assert derived.conclusion == goal
    assert derived.inference_rule is rule
    assert derived.premises == proofs


def test_unproved_presentation_or_zero_image_never_closes_root() -> None:
    goal, presentation = fixture_case()
    children = cyclic_generator_zero_image_inputs(goal, presentation)
    assert children is not None
~~~~

~~~~markdown
# Phase 174 R3-R7-R6: Cyclic Generator Zero-Image Audit

This package isolates a general inference for a finite-cyclic group presentation and a proved zero image of its generator. It does not register facts, change the production proof search, or declare a new completed group proof.

## Usage

From the repository root on Windows, extract the ZIP at the repository root and run:

```powershell
powershell -NoProfile -ExecutionPolicy Bypass -File ".\phase174_r3_r7_r6_cyclic_generator_zero_image\run.ps1" -DryRun
powershell -NoProfile -ExecutionPolicy Bypass -File ".\phase174_r3_r7_r6_cyclic_generator_zero_image\run.ps1"
```

The script runs focused pytest groups in separate processes because historical packages have identically named test modules. It never runs the full test suite.

The audit report is written to `%LOCALAPPDATA%\EHPProofTracerAudit\phase174_r3_r7_r6\cyclic_generator_zero_image.json`. Preconditions include a reviewed GitHub `develop` lineage, pinned source blobs, and the previous R5 live audit report.

The underlying Toda source is an adopted transcription; the original printed edition has not been independently authenticated. A registered database group presentation is an unproved proposal until its proper proof is reconstructed.

The package is deliberately **not** wired into the shared expansion provider or Web UI. A root candidate or a source quotation does not discharge its two independent proof obligations.
~~~~

~~~~markdown
## 出発点
~~~~
