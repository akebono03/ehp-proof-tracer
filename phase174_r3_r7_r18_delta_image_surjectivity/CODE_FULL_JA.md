"""Source-first, isolated Delta surjectivity from a verified full image.

A proof that im(Delta) is freely generated by g and a separate proof
that its codomain is freely generated by exactly the same typed g imply
surjectivity. This module proposes subgoals; it never asserts a GIVEN or
registers with the public search provider during R18.
"""
from __future__ import annotations

from homotopy_groups import (
    FreeCyclicGroup,
    TodaDeltaMap,
    TodaEHPExactnessWindow,
    TodaPrimaryGroup,
)
from map_facts import EHP_DELTA_MAP, EHP_E_MAP
from phase165_r12_r11_prop42_unified import prop42_exactness_indices
from phase165_r14_r6_generic_iso import MapSurjective
from phase165_r3_recursive_search import GoalExpansion
from proof import InferenceRule, PremisePattern, ProofRule, Relation, RelationType
from toda_rules import TodaDeltaImageFreeCyclicStatement, TodaProp42ExactnessStatement


RULE_NAME = "typed free cyclic Delta image equals codomain implies surjectivity"


def delta_surjectivity_image_inputs(
    goal: object, codomain_presentation: object,
) -> tuple[Relation, TodaDeltaImageFreeCyclicStatement] | None:
    """Build two separate typed obligations for a valid EHP Delta map."""
    if not isinstance(goal, MapSurjective) or not isinstance(goal.map, TodaDeltaMap):
        return None
    delta = goal.map
    source = delta.source_group
    codomain = delta.target_group
    if not isinstance(source, TodaPrimaryGroup) or not isinstance(codomain, TodaPrimaryGroup):
        return None
    q = codomain.group_dimension
    n = codomain.sphere_dimension
    if (type(q) is not int or type(n) is not int or n < 2 or q < n):
        return None
    following = TodaPrimaryGroup(q + 1, n + 1)
    exact = TodaProp42ExactnessStatement(TodaEHPExactnessWindow(
        source_term=source,
        middle_term=codomain,
        target_term=following,
        first_map=EHP_DELTA_MAP,
        second_map=EHP_E_MAP,
    ))
    if prop42_exactness_indices(exact) != (n, q + 1, "delta_e"):
        return None
    if (not isinstance(codomain_presentation, Relation)
            or codomain_presentation.relation_type is not RelationType.EQUALITY
            or codomain_presentation.lhs != codomain
            or not isinstance(codomain_presentation.rhs, FreeCyclicGroup)):
        return None
    generator = codomain_presentation.rhs.generator
    from expression import Expression, Zero
    if not isinstance(generator, Expression) or isinstance(generator, Zero):
        return None
    image = TodaDeltaImageFreeCyclicStatement(
        map=delta, image_group=FreeCyclicGroup(generator),
    )
    return codomain_presentation, image


def delta_surjectivity_from_full_free_cyclic_image_rule(
    goal: MapSurjective, codomain_presentation: Relation,
) -> InferenceRule:
    """Require two independently INFERRED, exactly matching premises."""
    required = delta_surjectivity_image_inputs(goal, codomain_presentation)
    if required is None:
        raise ValueError("invalid Delta surjectivity image/codomain obligations")

    def guard(premises, bindings):
        return (
            len(premises) == 2
            and all(step.rule is ProofRule.INFERENCE for step in premises)
            and tuple(step.conclusion for step in premises) == required
            and delta_surjectivity_image_inputs(goal, codomain_presentation) == required
        )

    return InferenceRule(
        name=RULE_NAME,
        description=(
            "If the independently proved image of a well-typed Delta map "
            "equals the independently proved free-cyclic codomain with "
            "the same exact typed generator, Delta is surjective. "
            "Neither the codomain structure nor its image is a GIVEN."
        ),
        premise_patterns=(
            PremisePattern(
                statement_type=Relation,
                relation_type=RelationType.EQUALITY,
                statement_pattern=required[0],
                proof_rule=ProofRule.INFERENCE,
            ),
            PremisePattern(
                statement_type=TodaDeltaImageFreeCyclicStatement,
                statement_pattern=required[1],
                proof_rule=ProofRule.INFERENCE,
            ),
        ),
        conclusion_pattern=goal,
        match_guard=guard,
    )


def expand_delta_surjectivity_from_full_free_cyclic_image(
    goal: object, codomain_presentation: object,
) -> tuple[GoalExpansion, ...]:
    """One isolated backward candidate; not a verified ProofStep."""
    required = delta_surjectivity_image_inputs(goal, codomain_presentation)
    if required is None:
        return ()
    return (GoalExpansion(
        rule=delta_surjectivity_from_full_free_cyclic_image_rule(
            goal, codomain_presentation,
        ),
        subgoals=required,
    ),)

def git(*args: str, root: Path = ROOT) -> str:
    return subprocess.run(
        ("git",) + args, cwd=root, check=True, capture_output=True,
        text=True, encoding="utf-8", errors="replace",
    ).stdout.strip()


def git_bytes(*args: str, root: Path = ROOT) -> bytes:
    return subprocess.run(
        ("git",) + args, cwd=root, check=True, capture_output=True,
    ).stdout


def git_blob_sha(data: bytes) -> str:
    return hashlib.sha1(
        b"blob " + str(len(data)).encode("ascii") + b"\0" + data,
    ).hexdigest()


def matches_git_blob(data: bytes, expected: str) -> bool:
    """Support uniform Windows CRLF checkout; reject partial/mixed EOLs."""
    if len(expected) != 40 or any(ch not in "0123456789abcdef" for ch in expected):
        return False
    remainder = data.replace(b"\r\n", b"")
    if b"\r" in remainder or (b"\r\n" in data and b"\n" in remainder):
        return False
    normalized = data.replace(b"\r\n", b"\n")
    return any(git_blob_sha(x) == expected for x in (
        data, normalized, normalized.replace(b"\n", b"\r\n"),
    ))


def checked_lineage(local: str, remote: str, changes: tuple[str, ...],
                    committed: tuple[str, ...]) -> str:
    """Permit exactly the two reviewed R17-to-R18 handoff states."""
    if remote != REVIEWED_HEAD or local not in (PRE_R17_HEAD, REVIEWED_HEAD):
        raise RuntimeError("Unreviewed GitHub/local develop HEAD; no checkout/reset")
    allowed_committed = set(HANDOFF_TRACKED)
    if (not set(HANDOFF_TRACKED).issubset(set(committed))
            or any(path not in allowed_committed and not path.startswith(
                ("phase174_r3_r7_r17_typed_map_open_goal/",
                 "phase174_r3_r7_r17r1_historical_test_contract/",
                 "phase174_r3_r7_r17r2_windows_git_path_guard_fix/"),
            ) for path in committed)):
        raise RuntimeError("Remote R17 merge lineage has unexpected files")
    if local == REVIEWED_HEAD and changes == ():
        return "R17_MERGED_CLEAN"
    if local == PRE_R17_HEAD and tuple(sorted(changes)) == HANDOFF_TRACKED:
        return "R17_PATCHES_UNCOMMITTED_MATCH_MERGED_REMOTE"
    raise RuntimeError("Unreviewed tracked worktree changes: " + repr(changes))


def verify_git_and_sources(root: Path = ROOT) -> dict[str, Any]:
    repo = Path(git("rev-parse", "--show-toplevel", root=root)).resolve()
    if repo != root.resolve():
        raise RuntimeError("Extract the R18 folder directly under repository root")
    if git("branch", "--show-current", root=root) != "develop":
        raise RuntimeError("R18 requires develop branch")
    local = git("rev-parse", "HEAD", root=root)
    remote_line = git("ls-remote", "origin", "refs/heads/develop", root=root)
    remote = remote_line.split()[0] if remote_line else ""
    if git("diff", "--cached", "--name-only", root=root):
        raise RuntimeError("Staged changes require separate audit")
    changes = tuple(git("diff", "--name-only", root=root).splitlines())
    git_bytes("merge-base", "--is-ancestor", PRE_R17_HEAD, REVIEWED_HEAD, root=root)
    commits = tuple(git("rev-list", PRE_R17_HEAD + ".." + REVIEWED_HEAD, root=root).splitlines())
    if len(commits) != 1:
        raise RuntimeError("Expected one reviewed R17 merge commit")
    changed_in_review = tuple(git(
        "diff", "--name-only", PRE_R17_HEAD, REVIEWED_HEAD, root=root,
    ).splitlines())
    lineage = checked_lineage(local, remote, changes, changed_in_review)
    for relative, blob in PINS.items():
        file = root / relative
        if not file.is_file() or not matches_git_blob(file.read_bytes(), blob):
            raise RuntimeError("R18 Source Guard mismatch: " + relative)
    for relative in HANDOFF_TRACKED:
        if not matches_git_blob(git_bytes(
                "show", "origin/develop:" + relative, root=root), PINS[relative]):
            raise RuntimeError("Reviewed GitHub R17 target mismatch: " + relative)
    return {
        "local_head": local, "remote_head": remote,
        "lineage": lineage, "source_guard": "PASS",
        "pinned_files": len(PINS), "tracked_changes": list(changes),
    }

def _inferred(statement: object) -> ProofStep:
    return ProofStep(
        conclusion=statement, premises=(), rule=ProofRule.INFERENCE,
        inference_rule=InferenceRule(name="test-only independent inferred fixture"),
    )


@pytest.mark.parametrize("q,n", ((11, 6), (9, 5), (13, 7), (17, 9)))

def test_only_two_matching_inferred_steps_can_prove_surjectivity() -> None:
    goal, group = _fixture(11, 6)
    obligations = delta_surjectivity_image_inputs(goal, group)
    assert obligations is not None
    rule = delta_surjectivity_from_full_free_cyclic_image_rule(goal, group)
