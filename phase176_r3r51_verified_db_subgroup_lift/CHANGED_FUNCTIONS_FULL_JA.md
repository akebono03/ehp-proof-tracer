```python
def finite_cyclic_subgroup_lift_rule(
    goal: CyclicStructureStatement, exactness: EHPImageEqualsKernelStatement,
    codomain: Relation, kernel: KernelMultipleStatement, image: Relation,
    *, bridge=None, proof_target=None,
) -> InferenceRule:
    """An injective H identifies its source with the proved subgroup <qb>."""
    required = finite_cyclic_subgroup_lift_inputs(goal, exactness, codomain, kernel, image)
    if required is None:
        raise ValueError("not a typed finite cyclic subgroup lift")

    def guard(premises, bindings):
        return (len(premises) == 5
                and all(premises[j].rule is ProofRule.INFERENCE for j in (0, 1, 3, 4))
                and (premises[2].rule is ProofRule.INFERENCE
                     or (bridge is not None and proof_target is not None
                         and verified_db_step(premises[2], proof_target=proof_target, bridge=bridge)))
                and tuple(step.conclusion for step in premises) == required
                and finite_cyclic_subgroup_lift_inputs(
                    goal, *(step.conclusion for step in premises[1:])) == required)

    return InferenceRule(
        name=RULE_NAME,
        description=("For injective H, exactness Im(H)=Ker(Delta), a cyclic "
                     "codomain of order m generated by b, Ker(Delta)=<qb>, "
                     "and H(a)=qb imply that a generates the source and has "
                     "order m/gcd(m,q). All five premises are independent."),
        premise_patterns=tuple(
            PremisePattern(proof_rule=(None if j == 2 and bridge is not None
                                       and proof_target is not None else ProofRule.INFERENCE),
                           statement_type=type(item), statement_pattern=item)
            for j, item in enumerate(required)
        ),
        conclusion_pattern=goal,
        match_guard=guard,
    )
```

```python
def expand_finite_cyclic_subgroup_lift(
    goal: object, exactness: object, codomain: object, kernel: object, image: object,
    *, bridge=None, proof_target=None,
) -> tuple[GoalExpansion, ...]:
    """Construct an isolated candidate from explicit typed witnesses only."""
    required = finite_cyclic_subgroup_lift_inputs(goal, exactness, codomain, kernel, image)
    if required is None:
        return ()
    return (GoalExpansion(
        finite_cyclic_subgroup_lift_rule(goal, exactness, codomain, kernel, image,
                                         bridge=bridge, proof_target=proof_target),
        required,
    ),)
```

```python
def build_public_subgroup_lift_provider(available: tuple, prior_conclusions: tuple = (),
                                        *, bridge=None, proof_target=None):
    """Source/prior candidates first, typed DB candidate last; no proof insertion."""
    if not isinstance(available, tuple) or not isinstance(prior_conclusions, tuple):
        raise TypeError("source and prior candidates must be tuples")

    def expand(goal):
        if (type(goal) is not CyclicStructureStatement
                or type(goal.group) is not TodaPrimaryGroup
                or type(goal.order) is not int or goal.order < 2):
            return ()
        a = goal.group
        i, n = a.group_dimension, a.sphere_dimension
        if type(i) is not int or type(n) is not int or n < 3 or i < 2*n-1:
            return ()
        b, c = TodaPrimaryGroup(i, 2*n-1), TodaPrimaryGroup(i-2, n-1)
        exact = EHPImageEqualsKernelStatement(TodaEHPExactnessWindow(a, b, c, EHP_H_MAP, EHP_DELTA_MAP))
        groups = []

        def add(item):
            if (type(item) is Relation and item.relation_type is RelationType.EQUALITY
                    and item.lhs == b and type(item.rhs) is FiniteCyclicGroup
                    and type(item.rhs.order) is int and item.rhs.order > goal.order
                    and item.rhs.order % goal.order == 0 and item not in groups):
                groups.append(item)

        for item in available + prior_conclusions:
            add(item)
            if type(item) is Relation:
                add(specialize_registered_higher_stem(item, b.sphere_dimension))
        # A DB presentation is a candidate for an OPEN codomain goal, never a
        # source leaf or a verified result. It can bind otherwise unknown b,m.
        for item in database_structured_group_goal_candidates(n=b.sphere_dimension,
                                                              k=i-b.sphere_dimension):
            add(item)
        options = []
        for group in groups:
            # Canonical generator of a subgroup of the requested order. The
            # equality H(a)=qb must still be proved; no unit/sign is assumed.
            q = group.rhs.order // goal.order
            generator = group.rhs.generator
            kernel = KernelMultipleStatement(TodaDeltaMap(b,c), b, generator, q)
            image = Relation(MapApplication(EHP_H_MAP, goal.generator), Multiple(q,generator), RelationType.EQUALITY)
            for option in expand_finite_cyclic_subgroup_lift(
                    goal, exact, group, kernel, image, bridge=bridge, proof_target=proof_target):
                if not any(old.subgoals == option.subgoals for old in options):
                    options.append(option)
        return tuple(options)

    return expand
```

```python
def updated_source(path, raw):
    text, eol = normalize(raw)
    if path == SHARED:
        # R3R51 changes only the existing R44 factory call. Accept that exact
        # successor without removing it; unknown subsequent edits still fail.
        from phase176_r3r51_verified_db_subgroup_lift.apply import OLD_HOOK, NEW_HOOK, PINS as R51_PINS
        from phase176_r3r46r1_prior_group_availability.apply import pinned
        if text.count(NEW_HOOK) == 1 and pinned(text.replace(NEW_HOOK, OLD_HOOK, 1), R51_PINS[SHARED]):
            return raw
        # Scope the source anchor to build_shared_typed_bridge. Other functions
        # may also create registries; the combined anchor is unique.
        anchor = '        sources = sources + (build_cited_toda55_quadruple_source(),)\n' + SOURCE_ANCHOR
        # Use a guarded insertion after the first line of that combined anchor.
        from phase176_r3r46r1_prior_group_availability.apply import pinned
        prefix = '        sources = sources + (build_cited_toda55_quadruple_source(),)\n'
        full = prefix + SOURCE_SETUP + SOURCE_ANCHOR
        provider_full = PROVIDER_SETUP + PROVIDER_ANCHOR
        base = text.replace(full, anchor, 1).replace(provider_full, PROVIDER_ANCHOR, 1)
        if not pinned(base, PINS[path]):
            raise ValueError('UNREVIEWED_SHARED_BLOB')
        if base.count(anchor) != 1 or base.count(PROVIDER_ANCHOR) != 1:
            raise ValueError('SHARED_ANCHOR_NOT_UNIQUE')
        changed = base.replace(anchor, full, 1).replace(PROVIDER_ANCHOR, provider_full, 1)
        if text not in (base, changed):
            raise ValueError('PARTIAL_OR_MOVED_PATCH')
        tree = ast.parse(changed)
        for name, token in (('build_shared_typed_bridge', 'build_cited_lemma57_delta_source'),
                            ('shared_expansion_provider', 'build_finite_delta_kernel_provider')):
            node = next(n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name == name)
            if token not in ast.get_source_segment(changed, node):
                raise ValueError('WRONG_FUNCTION_SCOPE:' + name)
    elif path == HISTORICAL_TEST:
        changed = checked_insertions(text, ((TEST_SETUP, TEST_ANCHOR),), PINS[path])
    else:
        raise ValueError('UNKNOWN_PATCH_TARGET')
    ast.parse(changed)
    return changed.replace('\n', eol).encode('utf-8')
```
